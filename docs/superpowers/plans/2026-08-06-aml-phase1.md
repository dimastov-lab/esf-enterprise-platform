# AML Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone AML service at `~/Projects/aml/` that receives alerts from external systems, lets analysts triage them, group them into cases, and record decisions (CLEARED / ESCALATED / SAR_FILED).

**Architecture:** Controller → Service → Repository → Database, with a GR Port stub (InMemoryGRAdapter) for customer snapshots. Two services (AlertService, CaseService) share the same FastAPI app and Postgres. Same patterns as Golden Record (`~/Projects/golden-record`).

**Tech Stack:** Python 3.12, FastAPI 0.115+, psycopg3 (AsyncConnectionPool), SQLAlchemy Core (schema only, no ORM), Alembic, Pydantic v2, pytest-asyncio, httpx

## Global Constraints

- Python ≥ 3.12; `from __future__ import annotations` in every file
- Package name: `aml`; env prefix: `AML_`; service port: `8200`; PostgreSQL port: `5435`
- psycopg3 autocommit by default; explicit `async with conn.transaction():` for multi-statement atomicity
- All repository Protocol methods must be `async def`
- Integration tests: function-scoped `AsyncConnectionPool`, unique `uuid4()` tenant per test, DELETE-based cleanup
- No ORM; raw SQL via `conn.execute(sql, params)` with `psycopg.rows.dict_row`
- RFC 7807 for all error responses; `map_exception()` in `api/errors.py`
- Bearer token via `hmac.compare_digest`; tenant from `AML_TENANT_ID` env var
- Alert state machine: `OPEN → IN_REVIEW → CLOSED` (forward only)
- Case state machine: `OPEN → UNDER_INVESTIGATION → CLOSED` (close only from UNDER_INVESTIGATION, outcome required)
- No TODO/FIXME/HACK in production code

```python
import uuid


class AlertNotFoundError(Exception):
    def __init__(self, alert_id: uuid.UUID) -> None:
        super().__init__(f"Alert {alert_id} not found.")
        self.alert_id = alert_id


class CaseNotFoundError(Exception):
    def __init__(self, case_id: uuid.UUID) -> None:
        super().__init__(f"Case {case_id} not found.")
        self.case_id = case_id


class GRCustomerNotFoundError(Exception):
    def __init__(self, customer_id: uuid.UUID) -> None:
        super().__init__(f"GR customer {customer_id} not found.")
        self.customer_id = customer_id


class InvalidTransitionError(Exception):
    def __init__(self, current: str, target: str) -> None:
        super().__init__(f"Cannot transition from {current!r} to {target!r}.")
        self.current = current
        self.target = target


class AlertAlreadyInCaseError(Exception):
    def __init__(self, alert_id: uuid.UUID, case_id: uuid.UUID) -> None:
        super().__init__(f"Alert {alert_id} is already in case {case_id}.")
        self.alert_id = alert_id
        self.case_id = case_id


class CaseClosedError(Exception):
    def __init__(self, case_id: uuid.UUID) -> None:
        super().__init__(f"Case {case_id} is already closed.")
        self.case_id = case_id
```

```python
from __future__ import annotations

from alembic import context
from sqlalchemy import engine_from_config, pool

from aml.db.schema import metadata

config = context.config
target_metadata = metadata


def run_migrations_offline() -> None:
    url = config.get_main_option("sqlalchemy.url")
    context.configure(url=url, target_metadata=target_metadata, literal_binds=True)
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    with connectable.connect() as connection:
        context.configure(connection=connection, target_metadata=target_metadata)
        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

```python
from typing import Any

import psycopg
import psycopg.rows
import psycopg_pool

from aml.domain.alert import Alert, AlertSeverity, AlertStatus


def _row_to_alert(row: dict[str, Any]) -> Alert:
    snap = row["customer_snapshot"]
    if isinstance(snap, str):
        snap = json.loads(snap)
    return Alert(
        id=uuid.UUID(str(row["id"])),
        tenant_id=uuid.UUID(str(row["tenant_id"])),
        gr_customer_id=uuid.UUID(str(row["gr_customer_id"])),
        customer_snapshot=snap,
        alert_type=row["alert_type"],
        description=row["description"],
        severity=AlertSeverity(row["severity"]),
        status=AlertStatus(row["status"]),
        source_system=row["source_system"],
        source_ref=row["source_ref"],
        created_at=row["created_at"],
        updated_at=row["updated_at"],
    )


class PgAlertRepository:
    def __init__(self, pool: psycopg_pool.AsyncConnectionPool) -> None:  # type: ignore[type-arg]
        self._pool = pool

    async def insert(self, alert: Alert) -> Alert:
        async with self._pool.connection() as conn:
            cur = await conn.execute(
                """
                INSERT INTO alerts
                  (id, tenant_id, gr_customer_id, customer_snapshot, alert_type,
                   description, severity, status, source_system, source_ref,
                   created_at, updated_at)
                VALUES (%s,%s,%s,%s::jsonb,%s,%s,%s,%s,%s,%s,%s,%s)
                RETURNING *
                """,
                (
                    str(alert.id), str(alert.tenant_id), str(alert.gr_customer_id),
                    json.dumps(alert.customer_snapshot), alert.alert_type, alert.description,
                    alert.severity.value, alert.status.value, alert.source_system,
                    alert.source_ref, alert.created_at, alert.updated_at,
                ),
            )
            cur.row_factory = psycopg.rows.dict_row  # type: ignore[assignment]
            row = await cur.fetchone()
            assert row is not None
            return _row_to_alert(row)

    async def get(self, tenant_id: uuid.UUID, alert_id: uuid.UUID) -> Alert | None:
        async with self._pool.connection() as conn:
            cur = await conn.execute(
                "SELECT * FROM alerts WHERE id = %s AND tenant_id = %s",
                (str(alert_id), str(tenant_id)),
            )
            cur.row_factory = psycopg.rows.dict_row  # type: ignore[assignment]
            row = await cur.fetchone()
            return _row_to_alert(row) if row else None

    async def list_alerts(
        self,
        tenant_id: uuid.UUID,
        *,
        status: AlertStatus | None = None,
        severity: AlertSeverity | None = None,
        gr_customer_id: uuid.UUID | None = None,
        limit: int = 20,
    ) -> list[Alert]:
        params: list[Any] = [str(tenant_id)]
        where = ["tenant_id = %s"]
        if status is not None:
            where.append("status = %s")
            params.append(status.value)
        if severity is not None:
            where.append("severity = %s")
            params.append(severity.value)
        if gr_customer_id is not None:
            where.append("gr_customer_id = %s")
            params.append(str(gr_customer_id))
        params.append(limit)
        sql = f"SELECT * FROM alerts WHERE {' AND '.join(where)} ORDER BY created_at LIMIT %s"
        async with self._pool.connection() as conn:
            cur = await conn.execute(sql, params)
            cur.row_factory = psycopg.rows.dict_row  # type: ignore[assignment]
            return [_row_to_alert(r) for r in await cur.fetchall()]

    async def update_status(
        self, tenant_id: uuid.UUID, alert_id: uuid.UUID, new_status: AlertStatus
    ) -> Alert | None:
        now = datetime.now(tz=timezone.utc)
        async with self._pool.connection() as conn:
            cur = await conn.execute(
                "UPDATE alerts SET status=%s, updated_at=%s WHERE id=%s AND tenant_id=%s RETURNING *",
                (new_status.value, now, str(alert_id), str(tenant_id)),
            )
            cur.row_factory = psycopg.rows.dict_row  # type: ignore[assignment]
            row = await cur.fetchone()
            return _row_to_alert(row) if row else None
```

```python
from datetime import datetime, timezone
```

```python
from aml.ports.gr_port import GRPort

_VALID: dict[AlertStatus, set[AlertStatus]] = {
    AlertStatus.OPEN: {AlertStatus.IN_REVIEW},
    AlertStatus.IN_REVIEW: {AlertStatus.CLOSED},
    AlertStatus.CLOSED: set(),
}


class AlertService:
    def __init__(self, gr: GRPort, repo: AlertRepository) -> None:
        self._gr = gr
        self._repo = repo

    async def create_alert(
        self,
        tenant_id: uuid.UUID,
        gr_customer_id: uuid.UUID,
        alert_type: str,
        description: str,
        severity: AlertSeverity,
        source_system: str,
        source_ref: str,
    ) -> Alert:
        snap = await self._gr.get_customer_snapshot(tenant_id, gr_customer_id)
        if snap is None:
            raise GRCustomerNotFoundError(gr_customer_id)
        now = datetime.now(tz=timezone.utc)
        return await self._repo.insert(Alert(
            id=uuid.uuid4(), tenant_id=tenant_id, gr_customer_id=gr_customer_id,
            customer_snapshot={"full_name": snap.full_name, "customer_type": snap.customer_type,
                               "tax_number": snap.tax_number},

            status=AlertStatus.OPEN, source_system=source_system, source_ref=source_ref,
            created_at=now, updated_at=now,
        ))

        self, tenant_id: uuid.UUID, alert_id: uuid.UUID, new_status: AlertStatus
    ) -> Alert:
        alert = await self._repo.get(tenant_id, alert_id)
        if alert is None:
            raise AlertNotFoundError(alert_id)
        if new_status not in _VALID[alert.status]:
            raise InvalidTransitionError(alert.status.value, new_status.value)

        alert = await self._repo.get(tenant_id, alert_id)
        if alert is None:
            raise AlertNotFoundError(alert_id)

        status: AlertStatus | None = None,
        severity: AlertSeverity | None = None,
        gr_customer_id: uuid.UUID | None = None,
```

```python
from __future__ import annotations

import uuid
from dataclasses import dataclass
from datetime import datetime, timezone

from aml.domain.alert import Alert
from aml.domain.case import Case, CaseOutcome, CaseStatus
from aml.domain.exceptions import (
    AlertAlreadyInCaseError, AlertNotFoundError,
    CaseClosedError, CaseNotFoundError, InvalidTransitionError,
)
from aml.ports.alert_repository import AlertRepository
from aml.ports.case_repository import CaseRepository

_VALID: dict[CaseStatus, set[CaseStatus]] = {
    CaseStatus.OPEN: {CaseStatus.UNDER_INVESTIGATION},
    CaseStatus.UNDER_INVESTIGATION: set(),
    CaseStatus.CLOSED: set(),
}


@dataclass(frozen=True, slots=True)
class CaseWithAlerts:
    case: Case
    alerts: list[Alert]


class CaseService:
    def __init__(self, case_repo: CaseRepository, alert_repo: AlertRepository) -> None:
        self._case_repo = case_repo
        self._alert_repo = alert_repo

    async def create_case(
        self, tenant_id: uuid.UUID, title: str, description: str, assigned_to: str | None = None
    ) -> Case:
        now = datetime.now(tz=timezone.utc)
        return await self._case_repo.insert(Case(
            id=uuid.uuid4(), tenant_id=tenant_id, title=title, description=description,
            status=CaseStatus.OPEN, outcome=None, assigned_to=assigned_to,
            created_at=now, updated_at=now, closed_at=None,
        ))

    async def get_case(self, tenant_id: uuid.UUID, case_id: uuid.UUID) -> CaseWithAlerts:
        result = await self._case_repo.get_with_alerts(tenant_id, case_id)
        if result is None:
            raise CaseNotFoundError(case_id)
        case, alerts = result
        return CaseWithAlerts(case=case, alerts=alerts)

    async def list_cases(
        self, tenant_id: uuid.UUID, *,
        status: CaseStatus | None = None,
        outcome: CaseOutcome | None = None,
        limit: int = 20,
    ) -> list[Case]:
        return await self._case_repo.list_cases(
            tenant_id, status=status, outcome=outcome, limit=limit
        )

    async def add_alert_to_case(
        self, tenant_id: uuid.UUID, case_id: uuid.UUID, alert_id: uuid.UUID
    ) -> CaseWithAlerts:
        case = await self._case_repo.get(tenant_id, case_id)
        if case is None:
            raise CaseNotFoundError(case_id)
        if case.status == CaseStatus.CLOSED:
            raise CaseClosedError(case_id)
        if await self._alert_repo.get(tenant_id, alert_id) is None:
            raise AlertNotFoundError(alert_id)
        if await self._case_repo.is_alert_in_case(case_id, alert_id):
            raise AlertAlreadyInCaseError(alert_id, case_id)
        await self._case_repo.add_alert(tenant_id, case_id, alert_id)
        return await self.get_case(tenant_id, case_id)

    async def remove_alert_from_case(
        self, tenant_id: uuid.UUID, case_id: uuid.UUID, alert_id: uuid.UUID
    ) -> None:
        if await self._case_repo.get(tenant_id, case_id) is None:
            raise CaseNotFoundError(case_id)
        if not await self._case_repo.remove_alert(tenant_id, case_id, alert_id):
            raise AlertNotFoundError(alert_id)

    async def transition_case(
        self, tenant_id: uuid.UUID, case_id: uuid.UUID, new_status: CaseStatus
    ) -> Case:
        case = await self._case_repo.get(tenant_id, case_id)
        if case is None:
            raise CaseNotFoundError(case_id)
        if new_status not in _VALID[case.status]:
            raise InvalidTransitionError(case.status.value, new_status.value)
        now = datetime.now(tz=timezone.utc)
        updated = Case(
            id=case.id, tenant_id=case.tenant_id, title=case.title,
            description=case.description, status=new_status, outcome=case.outcome,
            assigned_to=case.assigned_to, created_at=case.created_at,
            updated_at=now, closed_at=case.closed_at,
        )
        result = await self._case_repo.update(updated)
        assert result is not None
        return result

    async def close_case(
        self, tenant_id: uuid.UUID, case_id: uuid.UUID, outcome: CaseOutcome
    ) -> Case:
        case = await self._case_repo.get(tenant_id, case_id)
        if case is None:
            raise CaseNotFoundError(case_id)
        if case.status != CaseStatus.UNDER_INVESTIGATION:
            raise InvalidTransitionError(case.status.value, CaseStatus.CLOSED.value)
        now = datetime.now(tz=timezone.utc)
        updated = Case(
            id=case.id, tenant_id=case.tenant_id, title=case.title,
            description=case.description, status=CaseStatus.CLOSED, outcome=outcome,
            assigned_to=case.assigned_to, created_at=case.created_at,
            updated_at=now, closed_at=now,
        )
        result = await self._case_repo.update(updated)
        assert result is not None
        return result
```

```python
from __future__ import annotations

import hmac
import uuid

from aml.config import Settings


class AuthenticationError(Exception):
    pass


def require_auth(authorization: str | None, settings: Settings) -> uuid.UUID:
    if not authorization or not authorization.startswith("Bearer "):
        raise AuthenticationError()
    token = authorization.removeprefix("Bearer ").strip()
    if not hmac.compare_digest(token.encode(), settings.bearer_token.encode()):
        raise AuthenticationError()
    return settings.tenant_id
```

```python
def _body(status: int, title: str, detail: str) -> dict[str, Any]:
    return {
        "type": f"https://aml.internal/errors/{title}",
        "title": title,
        "status": status,
        "detail": detail,
    }


def map_exception(exc: Exception) -> tuple[int, dict[str, Any]]:
    if isinstance(exc, AuthenticationError):
        return 401, _body(401, "unauthorized", "Valid credentials are required.")
    if isinstance(exc, (AlertNotFoundError, CaseNotFoundError, GRCustomerNotFoundError)):
        return 404, _body(404, "not_found", str(exc))
    if isinstance(exc, (InvalidTransitionError, AlertAlreadyInCaseError, CaseClosedError)):
        return 409, _body(409, "conflict", str(exc))
    return 500, _body(500, "internal_error", "An internal error occurred.")
```
